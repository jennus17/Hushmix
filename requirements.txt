# Hushmix runtime dependencies
#
# Targets Python 3.10 - 3.12 (64-bit).  The old pin on pycaw 20181226 is rarely
# installable on current Python/Windows SDK combinations, so compatible ranges
# are used.  numpy/matplotlib were listed here but never imported by the
# application; the signal filtering is implemented in pure Python.

# --- Audio control (Windows Core Audio) -------------------------------------
pycaw>=20240210            # Core Audio bindings (AudioUtilities, IAudioEndpointVolume)
comtypes>=1.4.0            # COM support required by pycaw
pywin32>=306               # win32api / win32event / win32gui / win32process

# --- Hardware mixer ----------------------------------------------------------
pyserial>=3.5              # Serial communication with the mixer

# --- System utilities --------------------------------------------------------
psutil>=5.9                # Process lookup for the "current application" target

# --- GUI ---------------------------------------------------------------------
customtkinter>=5.2.0       # Themed Tk widgets
pystray>=0.19.4            # System tray icon
Pillow>=10.0.0             # Image handling for the tray icon

# --- Updates -----------------------------------------------------------------
requests>=2.31.0           # GitHub releases API and the update download
pefile>=2023.2.7           # Reads ProductVersion out of Hushmix.exe

# --- Button actions ----------------------------------------------------------
pyautogui>=0.9.54          # Keyboard shortcuts and media keys
